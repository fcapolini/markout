''}" // parameters :extra=${null} // extra classes for this instance :points=${[]} :stroke=${'var(--dash-accent)'} :fill=${true} :tall=${false} // private: the plot box, in the chart's own coordinates. A tall // chart is taller in its own units as well as in CSS, so the // stroke and the marker keep their proportions either way. // // The band the readings actually cover, not zero to peak: a // series that only moves between 99.4 and 100 is still a shape. :_h=${tall ? 90 : 30} :_top=${2} :_low=${points.length ? Math.min(...points) : 0} :_high=${points.length ? Math.max(...points) : 1} :_range=${Math.max(_high - _low, Math.abs(_high) / 1000, 1e-6)} :_xy=${points.map((v, i) => [ points.length < 2 ? 0 : (i / (points.length - 1)) * 100, _top + (1 - (v - _low) / _range) * (_h - _top * 2), ])} :_line=${_xy.map(p => `${p[0].toFixed(2)},${p[1].toFixed(2)}`).join(' ')} :_area=${`0,${_h} ${_line} 100,${_h}`} :_last=${_xy.length ? _xy[_xy.length - 1] : [0, 0]} // The marker is drawn as STROKE rather than as a shape. The box // is stretched to fill its element (`preserveAspectRatio="none"`) // and x and y are stretched by different amounts, so a // comes out as an ellipse — wider the wider the chart. A // zero-length path with a round cap is a dot whose size comes // from `stroke-width`, and `vector-effect="non-scaling-stroke"` // keeps that in device pixels, so it stays round at any size. :_dot=${`M ${_last[0].toFixed(2)} ${_last[1].toFixed(2)}` + ` L ${_last[0].toFixed(2)} ${_last[1].toFixed(2)}`} > <:define tag="dash-stat:div" class="card dash-stat h-100 ${extra ?? (delta == null ? 'secondary' : delta >= 0 ? 'success' : 'danger')} :_arrow=${delta == null ? '' : delta >= 0 ? '▲' : '▼'} >
${caption}
${_arrow} ${Math.abs(delta).toFixed(1)}%
