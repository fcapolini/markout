<:define tag="std-data:span" hidden // parameters :src="" // the URL; nothing is fetched without one :client=${false} // fetch in the browser instead of here // What the server fetched, and the only thing here that crosses. // Never rejects: a 404 is a fact about the page's data, not a // fault in the render, so it is carried back as a value the page // can show rather than reported to a log the visitor can't see. :server-_served=${ (client || !src) ? null : fetch(src) .then((res) => res.ok ? res.json().then((body) => ({ body })) : { error: `${res.status} ${res.statusText}` }) .catch((err) => ({ error: `${err}` })) } // what a browser fetch produced, if one has run :_fetched=${null} :_failed=${null} // the source of truth for readers. A browser fetch wins when there // has been one, since it is the newer of the two :data=${_fetched ?? _served?.body ?? null} :error=${_failed ?? _served?.error ?? null} :loading=${false} :reload=${() => { if (!src) { return; } loading = true; fetch(src) .then((res) => res.ok ? res.json().then((body) => { _fetched = body; _failed = null; }) : Promise.reject(new Error(`${res.status} ${res.statusText}`))) .catch((err) => { _failed = `${err}`; }) .then(() => { loading = false; }); }} // `:handle-` rather than `:did-init`, so a `:src` that changes // refetches instead of being read once. Browser-only, which is // what keeps this from running during the render; and it takes // the URL, so it fires again whenever that URL is a new one :_wanted=${client ? src : null} :handle-_wanted=${(v) => { v && reload(); }} />
