<:define tag="std-params:logic" :_search=${($url?.search ?? '').replace(/^\?/, '')} :_first=${_search.split(/[&;]/)[0]} // the page path, or nothing when the query opens with a parameter :_path=${_first.includes('=') ? '' : _first} // everything after it, which is the parameters and only them :_rest=${_path ? _search.slice(_path.length).replace(/^[&;]/, '') : _search} /* The parameters as a plain object, so a page reads `${q.data.id}` rather than calling a getter. A repeated name keeps its last value, which is what `Object.fromEntries` does with a duplicate key and what a page wanting every value should use `$url.searchParams.getAll` for. */ ::data=${Object.fromEntries(new URLSearchParams(_rest))} /* A URL for this page with `changes` merged into its parameters. Relative, so it keeps the pathname: the pathname is where a round trip lives, and a link that only changes the query is one this page answers itself. */ ::href=${(changes, page) => { const params = new URLSearchParams(_rest); Object.entries(changes ?? {}).forEach(([name, value]) => { value === null || value === undefined ? params.delete(name) : params.set(name, `${value}`); }); const path = page ?? _path; const rest = `${params}`; return `?${path}${path && rest ? '&' : ''}${rest}`; }} />
